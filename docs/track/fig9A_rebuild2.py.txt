import numpy as np
from math import sqrt, atan2, degrees, hypot, cos, sin, pi, radians

def circum(a, b, c):
    """Circle through 3 points -> (center, R)."""
    ax, ay = a; bx, by = b; cx, cy = c
    d = 2*(ax*(by-cy)+bx*(cy-ay)+cx*(ay-by))
    ux = ((ax**2+ay**2)*(by-cy)+(bx**2+by**2)*(cy-ay)+(cx**2+cy**2)*(ay-by))/d
    uy = ((ax**2+ay**2)*(cx-bx)+(bx**2+by**2)*(ax-cx)+(cx**2+cy**2)*(bx-ax))/d
    ctr = np.array([ux, uy])
    return ctr, hypot(ax-ux, ay-uy)

# ============ segments ============
class Line:
    def __init__(self, name, a, b, src):
        self.name, self.a, self.b, self.src, self.kind = name, np.asarray(a,float), np.asarray(b,float), src, "line"
    def start(self): return self.a
    def end(self):   return self.b
    def tang(self, p): return self.b - self.a

class Arc:
    def __init__(self, name, c, R, a0, a1, src, ccw=True):
        self.name, self.c, self.R, self.a0, self.a1, self.src, self.ccw = name, np.asarray(c,float), R, a0, a1, src, ccw
        self.kind = "arc"
    def pt(self, t): return self.c + self.R*np.array([cos(t), sin(t)])
    def start(self): return self.pt(self.a0)
    def end(self):   return self.pt(self.a1)
    def tang(self, p):
        r = p - self.c
        t = np.array([-r[1], r[0]])
        return t if self.ccw else -t

def ang(c, p): return atan2(p[1]-c[1], p[0]-c[0])

def gap(u, v):
    return hypot(*(np.asarray(u,float)-np.asarray(v,float)))

def check_ring(ring, label):
    print("="*100)
    print("[%s] adjacent-segment connection errors (auto-computed):" % label)
    n = len(ring)
    fails = 0
    for i in range(n):
        s, t = ring[i], ring[(i+1) % n]
        g = gap(s.end(), t.start())
        tag = "OK  " if g <= 0.05 else "FAIL"
        if g > 0.05: fails += 1
        print("  %-4s %-12s end -> %-12s start  gap=%.4f m" % (tag, s.name, t.name, g))
    print("  ==> %d FAIL(s)" % fails)
    print()
    print("[%s] tangent break angles at joints (deg; 0 = smooth):" % label)
    for i in range(n):
        s, t = ring[i], ring[(i+1) % n]
        d1 = s.tang(s.end());   d2 = t.tang(t.start())
        a = degrees(atan2(d2[1],d2[0]) - atan2(d1[1],d1[0])) % 180
        if a > 90: a = 180 - a
        print("  %-12s -> %-12s  tangent break = %6.2f deg" % (s.name, t.name, a))
    print()
