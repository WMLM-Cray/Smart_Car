def ang(c, p): return atan2(p[1]-c[1], p[0]-c[0])

class Line:
    def __init__(self, name, a, b, src):
        self.name, self.a, self.b, self.src, self.kind = name, np.asarray(a,float), np.asarray(b,float), src, "line"
    def start(self): return self.a
    def end(self): return self.b
    def tang(self, p): return self.b - self.a
    def samples(self, step=0.05):
        L = hypot(*(self.b-self.a)); n = max(int(L/step), 1)
        return [self.a + (self.b-self.a)*i/n for i in range(n+1)]

class Arc:
    def __init__(self, name, c, R, a0, a1, src, ccw=True):
        self.name, self.c, self.R, self.a0, self.a1, self.src, self.ccw = name, np.asarray(c,float), R, a0, a1, src, ccw
        self.kind = "arc"
        self.da = ((a1-a0) % (2*pi)) if ccw else ((a0-a1) % (2*pi))
    def pt(self, t): return self.c + self.R*np.array([cos(t), sin(t)])
    def start(self): return self.pt(self.a0)
    def end(self):   return self.pt(self.a1)
    def tang(self, p):
        r = p - self.c
        return np.array([-r[1], r[0]]) if self.ccw else np.array([r[1], -r[0]])
    def samples(self, step=0.05):
        n = max(int(self.da*self.R/step), 4)
        ts = [self.a0 + self.da*i/n for i in range(n+1)] if self.ccw else \
             [self.a0 - self.da*i/n for i in range(n+1)]
        return [self.pt(t) for t in ts]

# ============ checker ============
def wrap180(a):
    while a <= -180: a += 360
    while a > 180: a -= 360
    return a

def seg_crossings(A, B):
    a0, a1 = A[:-1], A[1:]
    b0, b1 = B[:-1], B[1:]
    def orient(p1, p2, q):
        return (p2[:,0]-p1[:,0])[:,None]*(q[None,:,1]-p1[:,None,1]) - \
               (p2[:,1]-p1[:,1])[:,None]*(q[None,:,0]-p1[:,None,0])
    o1 = orient(a0, a1, b0); o2 = orient(a0, a1, b1)
    o3 = orient(b0, b1, a0).T; o4 = orient(b0, b1, a1).T
    s1, s2, s3, s4 = np.sign(o1), np.sign(o2), np.sign(o3), np.sign(o4)
    straddle = ((s1*s2 <= 0) & (s3*s4 <= 0))
    collinear = (s1==0) & (s2==0) & (s3==0) & (s4==0)
    return int((straddle & ~collinear).sum())

def check_ring(ring, label, gap_tol=0.05, prox_tol=0.03):
    n = len(ring)
    fails = 0
    print("="*104)
    print("[%s] 1) connection gaps:" % label)
    for i in range(n):
        s, t = ring[i], ring[(i+1) % n]
        g = hypot(*(s.end() - t.start()))
        tag = "OK  " if g <= gap_tol else "FAIL"
        if g > gap_tol: fails += 1
        print("  %-4s %-14s -> %-14s gap=%.4f" % (tag, s.name, t.name, g))
    print("[%s] 2) DIRECTED tangent breaks (0=smooth, 180=backtrack):" % label)
    for i in range(n):
        s, t = ring[i], ring[(i+1) % n]
        d1 = s.tang(s.end()); d2 = t.tang(t.start())
        a = abs(wrap180(degrees(atan2(d2[1],d2[0]) - atan2(d1[1],d1[0]))))
        tag = ""
        if a > 170: tag = "  <-- BACKTRACK/REVERSAL"; fails += 1
        elif a > 0.5: tag = "  (corner)"
        print("  %-14s -> %-14s break=%7.2f deg%s" % (s.name, t.name, a, tag))
    print("[%s] 3) proximity screen (non-adjacent; SCREEN ONLY):" % label)
    samps = [np.array(s.samples(0.05)) for s in ring]
    bad = 0
    for i in range(n):
        for j in range(i+2, n):
            if i == 0 and j == n-1: continue
            dd = np.linalg.norm(samps[i][:,None,:]-samps[j][None,:,:], axis=2)
            m = dd.min()
            if m < prox_tol:
                print("  FAIL %-14s ~ %-14s min-dist=%.4f m" % (ring[i].name, ring[j].name, m))
                bad += 1
    if bad == 0:
        print("  none (all non-adjacent pairs keep >= %.2f m)" % prox_tol)
    fails += bad
    print("[%s] 4) self-intersection (segment crossing on sampled polylines):" % label)
    bad = 0
    for i in range(n):
        for j in range(i+2, n):
            if i == 0 and j == n-1: continue
            k = seg_crossings(samps[i], samps[j])
            if k > 0:
                print("  FAIL %-14s x %-14s %d crossing(s)" % (ring[i].name, ring[j].name, k))
                bad += 1
    if bad == 0:
        print("  none (no crossings among non-adjacent sampled segments)")
    fails += bad
    print("  ==> total FAIL(s): %d" % fails)
    print()
    return fails

def points_inside(pts, poly):
    """Ray casting: pts (m,2) against closed polyline poly (n,2). Returns bool array."""
    px, py = pts[:,0], pts[:,1]
    inside = np.zeros(len(pts), dtype=bool)
    x0, y0 = poly[:,0], poly[:,1]
    x1, y1 = np.roll(x0, -1), np.roll(y0, -1)
    for ax, ay, bx, by in zip(x0, y0, x1, y1):
        cond = ((ay > py) != (by > py))
        with np.errstate(divide='ignore', invalid='ignore'):
            xint = (bx-ax)*(py-ay)/(by-ay) + ax
        inside ^= (cond & (px < xint))
    return inside

def check_between(outer, inner, label="BETWEEN outer/inner"):
    """New: rings must not cross; inner ring strictly inside outer ring."""
    fails = 0
    print("="*104)
    print("[%s] 5) ring-pair checks:" % label)
    so = [np.array(s.samples(0.08)) for s in outer]
    si = [np.array(s.samples(0.08)) for s in inner]
    bad = 0
    for i, A in enumerate(so):
        for j, B in enumerate(si):
            k = seg_crossings(A, B)
            if k > 0:
                print("  FAIL outer %-13s x inner %-13s %d crossing(s)" % (outer[i].name, inner[j].name, k))
                bad += 1
    if bad == 0:
        print("  ring crossing: none (all outer x inner sampled segment pairs)")
    fails += bad
    inner_pts = np.vstack([a[::4] for a in si])      # coarse sample of inner ring
    outer_poly = np.vstack(so)
    ins = points_inside(inner_pts, outer_poly)
    nout = int((~ins).sum())
    if nout == 0:
        print("  inner ring inside outer ring: yes (%d points tested)" % len(inner_pts))
    else:
        print("  FAIL inner ring: %d/%d sampled points OUTSIDE outer ring" % (nout, len(inner_pts)))
        fails += 1
    dmin = min(np.linalg.norm(a[:,None,:]-b[None,:,:], axis=2).min()
               for a in so for b in si)
    print("  ring-pair min distance: %.3f m (informational)" % dmin)
    print("  ==> total FAIL(s): %d" % fails)
    print()
    return fails
