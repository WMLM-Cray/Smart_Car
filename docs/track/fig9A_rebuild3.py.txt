import numpy as np
from math import sqrt, atan2, degrees, hypot, cos, sin, pi, radians

def circum(a, b, c):
    ax, ay = a; bx, by = b; cx, cy = c
    d = 2*(ax*(by-cy)+bx*(cy-ay)+cx*(ay-by))
    ux = ((ax*ax+ay*ay)*(by-cy)+(bx*bx+by*by)*(cy-ay)+(cx*cx+cy*cy)*(ay-by))/d
    uy = ((ax*ax+ay*ay)*(cx-bx)+(bx*bx+by*by)*(ax-cx)+(cx*cx+cy*cy)*(bx-ax))/d
    return np.array([ux, uy]), hypot(ax-ux, ay-uy)

def pt(c, R, t): return c + R*np.array([cos(t), sin(t)])
def ang(c, p): return atan2(p[1]-c[1], p[0]-c[0])

class Line:
    def __init__(self, name, a, b, src):
        self.name, self.a, self.b, self.src, self.kind = name, np.asarray(a,float), np.asarray(b,float), src, "line"
    def start(self): return self.a
    def end(self): return self.b
    def tang(self, p): return self.b - self.a
    def samples(self, step=0.05):
        L = hypot(*(self.b-self.a)); n = max(int(L/step), 1)
        return [self.a + (self.b-self.a)*i/n for i in range(n+1)]

class Arc:
    def __init__(self, name, c, R, a0, a1, src, ccw=True):
        self.name, self.c, self.R, self.a0, self.a1, self.src, self.ccw = name, np.asarray(c,float), R, a0, a1, src, ccw
        self.kind = "arc"
        self.da = ((a1-a0) % (2*pi)) if ccw else ((a0-a1) % (2*pi))
    def pt(self, t): return self.c + self.R*np.array([cos(t), sin(t)])
    def start(self): return self.pt(self.a0)
    def end(self):   return self.pt(self.a1)
    def tang(self, p):
        r = p - self.c
        return np.array([-r[1], r[0]]) if self.ccw else np.array([r[1], -r[0]])
    def samples(self, step=0.05):
        n = max(int(self.da*self.R/step), 4)
        ts = [self.a0 + self.da*i/n for i in range(n+1)] if self.ccw else \
             [self.a0 - self.da*i/n for i in range(n+1)]
        return [self.pt(t) for t in ts]

# ============ checker ============
def wrap180(a):
    while a <= -180: a += 360
    while a > 180: a -= 360
    return a

def check_ring(ring, label, gap_tol=0.05, overlap_tol=0.03):
    n = len(ring)
    fails = 0
    print("="*104)
    print("[%s] 1) connection gaps (auto):" % label)
    for i in range(n):
        s, t = ring[i], ring[(i+1) % n]
        g = hypot(*(s.end() - t.start()))
        tag = "OK  " if g <= gap_tol else "FAIL"
        if g > gap_tol: fails += 1
        print("  %-4s %-13s -> %-13s gap=%.4f" % (tag, s.name, t.name, g))
    print("[%s] 2) DIRECTED tangent breaks (0=smooth, 180=backtrack):" % label)
    for i in range(n):
        s, t = ring[i], ring[(i+1) % n]
        d1 = s.tang(s.end()); d2 = t.tang(t.start())
        a = abs(wrap180(degrees(atan2(d2[1],d2[0]) - atan2(d1[1],d1[0]))))
        tag = ""
        if a > 170: tag = "  <-- BACKTRACK/REVERSAL"; fails += 1
        elif a > 0.5: tag = "  (corner)"
        print("  %-13s -> %-13s break=%7.2f deg%s" % (s.name, t.name, a, tag))
    print("[%s] 3) non-adjacent overlap / self-intersection:" % label)
    samps = [np.array(s.samples()) for s in ring]
    bad = 0
    for i in range(n):
        for j in range(i+2, n):
            if i == 0 and j == n-1: continue
            A, B = samps[i], samps[j]
            # min pairwise distance (rings are small; fine)
            dd = np.linalg.norm(A[:,None,:]-B[None,:,:], axis=2)
            m = dd.min()
            if m < overlap_tol:
                print("  FAIL %-13s ~ %-13s min-dist=%.4f m" % (ring[i].name, ring[j].name, m))
                bad += 1
    if bad == 0:
        print("  none (all non-adjacent pairs keep >= %.2f m)" % overlap_tol)
    fails += bad
    print("  ==> total FAIL(s): %d" % fails)
    print()
    return fails

# self-test: deliberately broken ring must be caught (proves checker works)
def selftest():
    print("="*104)
    print("[SELFTEST] broken ring: old v2 bottom (16.8,0)->(0,0) then (0,0)->(2.75,0) backtrack")
    bad = [
        Line("B1", [16.8,0.0], [0.0,0.0], ""),
        Line("B2", [0.0,0.0], LL_OBOT, ""),
    ]
    d1 = bad[0].tang(bad[0].end()); d2 = bad[1].tang(bad[1].start())
    a = abs(wrap180(degrees(atan2(d2[1],d2[0]) - atan2(d1[1],d1[0]))))
    A = np.array(bad[0].samples()); B = np.array(bad[1].samples())
    m = np.linalg.norm(A[:,None,:]-B[None,:,:], axis=2).min()
    print("  directed break at joint = %.2f deg (expect 180) -> %s" % (a, "CAUGHT" if a > 170 else "MISSED"))
    print("  overlap min-dist = %.4f m (expect 0) -> %s" % (m, "CAUGHT" if m < 0.03 else "MISSED"))
    print()
