# dilate masks by ~2px so "distance to nearest stroke" <= 2px counts as 0
def dilate(m, it=2):
    r = m.copy()
    for _ in range(it):
        r2 = r.copy()
        r2[1:, :] |= r[:-1, :]; r2[:-1, :] |= r[1:, :]
        r2[:, 1:] |= r[:, :-1]; r2[:, :-1] |= r[:, 1:]
        r = r2
    return r
mBd, mRd, mYd = dilate(mB), dilate(mR), dilate(mY)

def to_px(xm, ym):
    return xm / s + bxmin, YM0 - ym / s

def line_pts(p, q, step=0.1):
    L = math.hypot(q[0] - p[0], q[1] - p[1])
    n = max(2, int(L / step) + 1)
    return [(p[0] + (q[0] - p[0]) * i / (n - 1), p[1] + (q[1] - p[1]) * i / (n - 1)) for i in range(n)]

def arc_pts(c, r, a1, a2, step=0.1):  # angles deg, sweep a1->a2
    da = (a2 - a1) % 360
    if da > 180: da -= 360
    L = abs(da) / 180 * math.pi * r
    n = max(2, int(L / step) + 1)
    return [(c[0] + r * math.cos(math.radians(a1 + da * i / (n - 1))),
             c[1] + r * math.sin(math.radians(a1 + da * i / (n - 1)))) for i in range(n)]

# ---------------- deviation ----------------
def dev(pts, mask):
    ds = []
    for xm, ym in pts:
        X, Y = to_px(xm, ym)
        X, Y = int(round(X)), int(round(Y))
        if not (0 <= X < W and 0 <= Y < H):
            ds.append(None); continue
        if mask[Y, X]:
            ds.append(0.0); continue
        best = None
        for rad in range(1, 9):
            found = False
            x0, x1 = max(0, X - rad), min(W, X + rad + 1)
            y0, y1 = max(0, Y - rad), min(H, Y + rad + 1)
            if mask[y0:y1, x0:x1].any():
                best = rad; found = True
            # ring only: cheap approximation -> recompute precisely below
            if found:
                break
        if best is None:
            ds.append(None); continue
        # exact nearest within window
        x0, x1 = max(0, X - 9), min(W, X + 10)
        y0, y1 = max(0, Y - 9), min(H, Y + 10)
        sub = mask[y0:y1, x0:x1]
        ys, xs = np.nonzero(sub)
        d = np.min(np.hypot(xs + x0 - X, ys + y0 - Y))
        ds.append(d * s)
    dd = [x for x in ds if x is not None]
    cov = len(dd) / len(ds)
    if not dd:
        return 0.0, 99.0, 0.0
    return sum(dd) / len(dd), max(dd), cov

    if len(pts) < 2:

    m, mx, cv = dev(pts, mask)
    print("%-4s %-5s %-28s %8.3f %8.3f %6.2f" % (sid, kind, note, m, mx, cv))
