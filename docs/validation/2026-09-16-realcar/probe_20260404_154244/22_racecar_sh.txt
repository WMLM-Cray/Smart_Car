#!/bin/bash

Run_car=""
Run_car_pgid=""
slam_gmapping_pid=""
slam_gmapping_pgid=""
terminating=0

terminate() {
  if [ "$terminating" -eq 1 ]; then
    return
  fi
  terminating=1

  echo "终止所有后台进程..."

  for pid in "$Run_car" "$slam_gmapping_pid"; do
    if [ -n "$pid" ] && kill -0 "$pid" 2>/dev/null; then
      kill "$pid" 2>/dev/null
    fi
  done

  for pgid in "$Run_car_pgid" "$slam_gmapping_pgid"; do
    if [ -n "$pgid" ]; then
      kill -TERM -- "-$pgid" 2>/dev/null
    fi
  done

  # Fallback cleanup for any lingering ROS 2 processes started by these launches
  pkill -f "ros2 launch racecar Run_car.launch.py" 2>/dev/null || true
  pkill -f "ros2 launch slam_gmapping slam_gmapping.launch.py" 2>/dev/null || true
  pkill -f "/slam_gmapping" 2>/dev/null || true
  pkill -f "hipnuc_imu/talker" 2>/dev/null || true

  wait "$Run_car" "$slam_gmapping_pid" 2>/dev/null
  echo "所有进程已终止。"
}

trap terminate SIGINT SIGTERM EXIT


# 启动 ROS 2 launch 文件

echo "启动 ROS 2 launch 文件: Run_car.launch.py"
setsid ros2 launch racecar Run_car.launch.py &
Run_car=$!
Run_car_pgid=$Run_car

# 等待5秒
sleep 5
