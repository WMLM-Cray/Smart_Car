cmake_minimum_required(VERSION 3.8)
project(shadow_feed)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(shadow_feed src/shadow_feed.cpp)
target_compile_options(shadow_feed PRIVATE -O2)
ament_target_dependencies(shadow_feed rclcpp sensor_msgs nav_msgs std_msgs)
install(TARGETS shadow_feed DESTINATION lib/${PROJECT_NAME})
ament_package()
